cmake_minimum_required(VERSION 3.19)
project(Imx6ullVehicleTerminal LANGUAGES C CXX)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(CTest)
option(VEHICLE_BUILD_CLIENT "Build the Qt desktop client" OFF)
option(VEHICLE_BUILD_SERVER "Build the Linux TCP service" ${UNIX})
if(VEHICLE_BUILD_SERVER AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/server/CMakeLists.txt")
    add_subdirectory(server)
endif()
if(BUILD_TESTING AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/CMakeLists.txt")
    add_subdirectory(tests)
endif()
