cmake_minimum_required(VERSION 3.19)
project(Imx6ullVehicleTerminal LANGUAGES C CXX)

set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

include(CTest)
option(VEHICLE_BUILD_CLIENT "Build the Qt desktop client" ON)
option(VEHICLE_BUILD_SERVER "Build the Linux TCP service" ${UNIX})

if(VEHICLE_BUILD_CLIENT OR BUILD_TESTING)
    find_package(Qt6 6.5 REQUIRED COMPONENTS Core Widgets Network)
    qt_standard_project_setup()
endif()

if(VEHICLE_BUILD_CLIENT)
    add_subdirectory(client)
endif()
if(VEHICLE_BUILD_SERVER)
    if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
        message(FATAL_ERROR "The TCP service requires Linux/V4L2. Use -DVEHICLE_BUILD_SERVER=OFF on this platform.")
    endif()
    add_subdirectory(server)
endif()
if(BUILD_TESTING)
    add_subdirectory(tests)
endif()
